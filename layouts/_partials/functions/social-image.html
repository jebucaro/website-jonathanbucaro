{{- /* Returns a dict {url, width, height, type} for og:image / twitter:image. Source priority: page image → hero image (home) → site logo. Social platforms (LinkedIn, X, Facebook) expect a landscape image around 1200x630 (1.91:1) and hard-crop anything else from the center, which cuts off card/hero art that's square or portrait. So when the source has enough resolution to crop cleanly, this smart-crops it to exactly 1200x630 as a JPEG (broadest crawler compatibility). Sources too small to crop without heavy upscaling (e.g. the site logo) keep their native aspect ratio instead, same as before. */ -}} {{- $path := "" -}} {{- $res := "" -}} {{- if .IsHome -}} {{- $path = .Site.Params.hero.hero__image -}} {{- $res = resources.Get (strings.TrimPrefix "/" $path) -}} {{- else if .Params.image -}} {{- $path = .Params.image -}} {{- $res = .Resources.GetMatch .Params.image -}} {{- else -}} {{- $path = .Site.Params.logo -}} {{- $res = resources.Get (strings.TrimPrefix "/" $path) -}} {{- end -}} {{- $url := $path | absURL -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- $type := "" -}} {{- with $res -}} {{- if in (slice "png" "jpeg" "webp" "tiff" "bmp") .MediaType.SubType -}} {{- if and (ge .Width 600) (ge .Height 600) -}} {{- $social := .Fill "1200x630 jpg q75 Smart" -}} {{- $url = $social.Permalink -}} {{- $width = $social.Width -}} {{- $height = $social.Height -}} {{- $type = $social.MediaType.Type -}} {{- else -}} {{- if gt .Width 1200 -}} {{- $capped := .Resize "1200x" -}} {{- $url = $capped.Permalink -}} {{- $width = $capped.Width -}} {{- $height = $capped.Height -}} {{- else -}} {{- $url = .Permalink -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- end -}} {{- $type = .MediaType.Type -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "url" $url "width" $width "height" $height "type" $type) -}}
