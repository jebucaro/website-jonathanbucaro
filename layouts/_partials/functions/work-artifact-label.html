{{- /* Resolves a work item's `artifact` slug to its localized label, or "" when the item declares none. Callers must render the result as plain text: `artifact` is not a taxonomy, so it has no term page to link to. Also the only reader of `.Params.artifact`, so it validates the slug against `work.artifacts` in config/_default/params.yaml and fails the build otherwise. Usage: {{ $artifact := partial "functions/work-artifact-label.html" (dict "page" .) }} {{ with $artifact }}{{ . }}{{ end }} */ -}} {{- $page := .page -}} {{- $slug := $page.Params.artifact -}} {{- $label := "" -}} {{- with $slug -}} {{- $allowed := site.Params.work.artifacts | default slice -}} {{- if not (in $allowed .) -}} {{- errorf "work-artifact-label: unknown artifact %q on %q. Allowed values (work.artifacts in config/_default/params.yaml): %v" . $page.Path $allowed -}} {{- else -}} {{- $label = i18n (printf "work_artifact_%s" .) -}} {{- end -}} {{- end -}} {{- return $label -}}
